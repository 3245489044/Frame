### 1.4 模块管理调用格式

```csharp
// 模块管理调用函数格式
public class ModuleManager
{
    private readonly IModuleRegistry _moduleRegistry;

    public ModuleManager(IModuleRegistry moduleRegistry)
    {
        _moduleRegistry = moduleRegistry;
    }

    // 注册模块函数
    public async Task RegisterModuleAsync(ModuleInfo moduleInfo)
    {
        // 注册模块信息
        await _moduleRegistry.RegisterModuleAsync(moduleInfo);
    }

    // 注销模块函数
    public async Task UnregisterModuleAsync(string moduleId)
    {
        // 注销指定模块
        await _moduleRegistry.UnregisterModuleAsync(moduleId);
    }

    // 获取模块信息函数
    public async Task<ModuleInfo> GetModuleAsync(string moduleId)
    {
        // 获取指定模块信息
        return await _moduleRegistry.GetModuleAsync(moduleId);
    }

    // 加载模块函数
    public async Task LoadModuleAsync(string moduleId)
    {
        // 加载指定模块
        await _moduleRegistry.LoadModuleAsync(moduleId);
    }

    // 卸载模块函数
    public async Task UnloadModuleAsync(string moduleId)
    {
        // 卸载指定模块
        await _moduleRegistry.UnloadModuleAsync(moduleId);
    }

    // 发现模块函数
    public async Task DiscoverModulesAsync(string searchPath)
    {
        // 发现指定路径下的模块
        await _moduleRegistry.DiscoverModulesAsync(searchPath);
    }

    // 获取所有模块函数
    public async Task<IEnumerable<ModuleInfo>> GetAllModulesAsync()
    {
        // 获取所有可用模块
        return await _moduleRegistry.GetAllModulesAsync();
    }
}
```

## 2. 模块子项目构建指南

### 2.1 项目创建步骤

#### 2.1.1 创建新项目

```bash
# 创建控制台应用
 dotnet new console -n MyModuleProject -o MyModuleProject

# 或创建类库项目
 dotnet new classlib -n MyModuleLibrary -o MyModuleLibrary
```

#### 2.1.2 添加项目依赖

在项目的 `.csproj` 文件中添加以下依赖：

```xml
<Project Sdk="Microsoft.NET.Sdk">

  <PropertyGroup>
    <OutputType>Exe</OutputType>
    <TargetFramework>net10.0</TargetFramework>
    <ImplicitUsings>enable</ImplicitUsings>
    <Nullable>enable</Nullable>
  </PropertyGroup>

  <ItemGroup>
    <!-- 添加FrameManage项目引用 -->
    <ProjectReference Include="..\FrameManage\FrameManage.csproj" />
  </ItemGroup>

  <ItemGroup>
    <!-- 添加必要的NuGet包 -->
    <PackageReference Include="Microsoft.Extensions.DependencyInjection" Version="10.0.6" />
    <PackageReference Include="Microsoft.Extensions.Configuration" Version="8.0.0" />
    <PackageReference Include="Microsoft.Extensions.Configuration.Json" Version="8.0.1" />
  </ItemGroup>

</Project>
```

### 2.2 项目文件结构

推荐的模块子项目文件结构：

```
MyModuleProject/
├── Properties/
│   └── launchSettings.json
├── Workflows/           # 工作流定义
│   └── UserRegistrationWorkflow.cs
├── DataModels/          # 数据模型
│   └── UserModel.cs
├── Services/            # 业务服务
│   └── UserService.cs
├── Program.cs           # 应用入口
├── MyModuleProject.csproj
└── appsettings.json     # 配置文件
```

### 2.3 单文件构建方案

对于快速测试和简单场景，您可以使用单文件方式构建完整的子模块项目。以下是单文件构建的步骤：

#### 2.3.1 创建单文件项目

```bash
# 创建单文件控制台应用
dotnet new console -n SingleFileModule -o SingleFileModule
cd SingleFileModule
```

#### 2.3.2 修改项目文件

在 `SingleFileModule.csproj` 文件中添加以下内容：

```xml
<Project Sdk="Microsoft.NET.Sdk">

  <PropertyGroup>
    <OutputType>Exe</OutputType>
    <TargetFramework>net10.0</TargetFramework>
    <ImplicitUsings>enable</ImplicitUsings>
    <Nullable>enable</Nullable>
  </PropertyGroup>

  <ItemGroup>
    <!-- 添加FrameManage项目引用 -->
    <ProjectReference Include="..\FrameManage\FrameManage.csproj" />
  </ItemGroup>

  <ItemGroup>
    <!-- 添加必要的NuGet包 -->
    <PackageReference Include="Microsoft.Extensions.DependencyInjection" Version="10.0.6" />
    <PackageReference Include="Microsoft.Extensions.Configuration" Version="8.0.0" />
    <PackageReference Include="Microsoft.Extensions.Configuration.Json" Version="8.0.1" />
  </ItemGroup>

</Project>
```

#### 2.3.3 创建单文件Program.cs

在 `Program.cs` 文件中添加完整的模块代码。单文件构建的代码与 [2.4 完整的模块项目示例](#24-完整的模块项目示例) 中的代码逻辑相同，只是将所有代码合并到一个文件中。

单文件版本的主要特点：
- 将数据模型、业务服务、工作流定义和主程序放在同一个文件中
- 命名空间使用 `SingleFileModule` 而不是 `MyModuleProject`
- 适合快速测试和简单场景

#### 2.3.4 构建和运行单文件项目

```bash
# 构建项目
dotnet build

# 运行项目
dotnet run
```

namespace MyModuleProject.Services

#### 2.4.4 工作流定义 (UserRegistrationWorkflow.cs)

```csharp
using FrameManage.MetaWorkflow;
using System.Collections.Generic;

namespace MyModuleProject.Workflows
{
    /// <summary>
    /// 用户注册工作流定义类
    /// 用于创建和配置用户注册流程的元工作流
    /// </summary>
    public class UserRegistrationWorkflow
    {
        /// <summary>
        /// 创建用户注册工作流定义
        /// </summary>
        /// <returns>返回配置好的元工作流定义对象</returns>
        public static MetaWorkflowDefinition CreateWorkflow()
        {
            // 创建元工作流定义对象
            return new MetaWorkflowDefinition
            {
                // 工作流名称
                Name = "用户注册工作流",
                // 工作流描述
                Description = "处理用户注册流程",
                // 工作流节点列表
                Nodes = new List<MetaWorkflowNode>
                {
                    // 第一个节点：验证节点
                    new MetaWorkflowNode
                    {
                        // 节点ID，用于唯一标识节点
                        Id = "validate",
                        // 节点名称，用于显示和描述
                        Name = "验证节点",
                        // 框架名称，指定使用哪个框架执行此节点
                        FrameworkName = "FrameManage",
                        // 工作流ID，指定要执行的具体工作流
                        WorkflowId = "validate_user_input"
                        // 注意：第一个节点不需要指定PreviousNodeIds
                    },
                    // 第二个节点：创建节点
                    new MetaWorkflowNode
                    {
                        Id = "create",
                        Name = "创建节点",
                        FrameworkName = "FrameManage",
                        WorkflowId = "create_user",
                        // 指定前置节点ID，表示此节点依赖于validate节点
                        PreviousNodeIds = new List<string> { "validate" }
                    },
                    // 第三个节点：通知节点
                    new MetaWorkflowNode
                    {
                        Id = "notify",
                        Name = "通知节点",
                        FrameworkName = "FrameManage",
                        WorkflowId = "send_confirmation_email",
                        // 指定前置节点ID，表示此节点依赖于create节点
                        PreviousNodeIds = new List<string> { "create" }
                    }
                }
            };
        }
    }
}
```

**说明：**
- 此代码定义了一个用户注册工作流，包含三个节点：验证节点、创建节点和通知节点
- 每个节点都指定了唯一的ID、名称、使用的框架和要执行的工作流
- 通过PreviousNodeIds属性指定节点之间的依赖关系，形成工作流的执行顺序
- 工作流执行时会按照节点之间的依赖关系顺序执行：validate → create → notify

#### 2.4.5 主程序 (Program.cs)

```csharp
using System;
using System.Collections.Generic;
using System.Threading.Tasks;
using FrameManage.Core;
using MyModuleProject.DataModels;
using MyModuleProject.Services;
using MyModuleProject.Workflows;

namespace MyModuleProject
{
    /// <summary>
    /// 应用程序主类
    /// 用于演示 FrameManage 模块项目的使用方法
    /// </summary>
    class Program
    {
        /// <summary>
        /// 应用程序入口点
        /// </summary>
        /// <param name="args">命令行参数</param>
        /// <returns>异步任务</returns>
        static async Task Main(string[] args)
        {
            Console.WriteLine("=== FrameManage 模块项目示例 ===");

            // 1. 初始化应用程序
            // 创建 Application 实例并调用 InitializeAsync 方法进行初始化
            var application = new Application();
            await application.InitializeAsync();

            // 2. 注册工作流函数
            // 创建 UserService 实例，用于处理用户相关的业务逻辑
            var userService = new UserService();

            // 注册验证用户输入工作流
            // 工作流ID: validate_user_input
            // 功能: 验证用户输入数据的有效性
            await application.RegisterWorkflowAsync("validate_user_input", async input =>
            {
                // 检查输入数据格式是否正确
                if (input is Dictionary<string, object> inputDict && inputDict.ContainsKey("user_data"))
                {
                    var userData = inputDict["user_data"] as Dictionary<string, object>;
                    if (userData != null)
                    {
                        // 从输入数据创建 UserModel 对象
                        var user = new UserModel
                        {
                            Name = userData.ContainsKey("name") ? userData["name"] as string : string.Empty,
                            Email = userData.ContainsKey("email") ? userData["email"] as string : string.Empty,
                            Password = userData.ContainsKey("password") ? userData["password"] as string : string.Empty
                        };
                        // 调用 UserService 的 ValidateUserInputAsync 方法验证用户输入
                        return await userService.ValidateUserInputAsync(user);
                    }
                }
                // 输入数据格式不正确，返回 false
                return false;
            });

            // 注册创建用户工作流
            // 工作流ID: create_user
            // 功能: 创建新用户并返回用户ID
            await application.RegisterWorkflowAsync("create_user", async input =>
            {
                // 检查输入数据格式是否正确
                if (input is Dictionary<string, object> inputDict && inputDict.ContainsKey("user_data"))
                {
                    var userData = inputDict["user_data"] as Dictionary<string, object>;
                    if (userData != null)
                    {
                        // 从输入数据创建 UserModel 对象
                        var user = new UserModel
                        {
                            Name = userData.ContainsKey("name") ? userData["name"] as string : string.Empty,
                            Email = userData.ContainsKey("email") ? userData["email"] as string : string.Empty,
                            Password = userData.ContainsKey("password") ? userData["password"] as string : string.Empty
                        };
                        // 调用 UserService 的 CreateUserAsync 方法创建用户
                        var createdUser = await userService.CreateUserAsync(user);
                        // 返回包含用户ID的字典
                        return new Dictionary<string, object> { { "user_id", createdUser.Id } };
                    }
                }
                // 输入数据格式不正确，返回空用户ID
                return new Dictionary<string, object> { { "user_id", string.Empty } };
            });

            // 注册发送确认邮件工作流
            // 工作流ID: send_confirmation_email
            // 功能: 向用户发送确认邮件
            await application.RegisterWorkflowAsync("send_confirmation_email", async input =>
            {
                // 检查输入数据格式是否正确
                if (input is Dictionary<string, object> inputDict && inputDict.ContainsKey("user_id"))
                {
                    // 获取用户ID
                    var userId = inputDict["user_id"] as string;
                    // 调用 UserService 的 SendConfirmationEmailAsync 方法发送确认邮件
                    return await userService.SendConfirmationEmailAsync(userId);
                }
                // 输入数据格式不正确，返回 false
                return false;
            });

            // 3. 注册元工作流
            // 创建用户注册元工作流定义
            var metaWorkflow = UserRegistrationWorkflow.CreateWorkflow();
            // 注册元工作流，指定工作流ID为 user_registration_meta
            await application.RegisterMetaWorkflowAsync("user_registration_meta", metaWorkflow);

            // 4. 执行元工作流
            Console.WriteLine("\n执行用户注册元工作流...");
            // 准备输入数据
            var input = new Dictionary<string, object>
            {
                { "user_data", new Dictionary<string, object>
                    {
                        { "name", "John Doe" },
                        { "email", "john.doe@example.com" },
                        { "password", "Secure123!" }
                    }
                }
            };

            // 执行元工作流并获取结果
            var result = await application.ExecuteMetaWorkflowAsync("user_registration_meta", input);
            Console.WriteLine($"\n工作流执行结果: {result}");

            // 5. 获取所有工作流
            Console.WriteLine("\n获取所有可用工作流:");
            var workflows = await application.GetAllWorkflowsAsync();
            foreach (var workflow in workflows)
            {
                Console.WriteLine($"- {workflow}");
            }

            Console.WriteLine("\n按任意键退出...");
            Console.ReadKey();
        }
    }
}
```

**说明：**
- 此代码演示了如何使用 FrameManage 框架创建和执行一个完整的用户注册工作流
- 主要步骤包括：
  1. 初始化应用程序
  2. 注册三个基础工作流：验证用户输入、创建用户、发送确认邮件
  3. 注册元工作流，将三个基础工作流组织成一个完整的流程
  4. 执行元工作流，处理用户注册请求
  5. 获取并显示所有可用的工作流
- 工作流执行顺序：validate_user_input → create_user → send_confirmation_email
- 每个工作流都有明确的职责，通过元工作流组织成一个完整的业务流程
