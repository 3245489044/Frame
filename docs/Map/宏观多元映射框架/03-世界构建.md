## 2. 空间离散化

### 2.1 数学定义

```
连续空间 ℝ³ 到离散网格 ℤ³ 的映射：

  G: ℝ³ → ℤ³
  G(x, y, z) = (floor(x / R), floor(y / R), floor(z / R))

其中 R 是体素分辨率（每个体素的边长，单位：米/体素）

逆映射（体素中心到连续空间）：
  G⁻¹(i, j, k) = ((i + 0.5) · R, (j + 0.5) · R, (k + 0.5) · R)
```

### 2.2 分辨率定义

```
分辨率 R 决定了一个体素在物理世界中的大小：

  R = 0.01m  →  1cm³ 体素（高精度）
  R = 0.1m   →  10cm³ 体素（中等精度）
  R = 1.0m   →  1m³ 体素（低精度）

体素坐标 (i,j,k) 与物理坐标 (x,y,z) 的转换：
  体素坐标 = floor(物理坐标 / R)
  物理坐标 = (体素坐标 + 0.5) · R
```

### 2.3 编程实现

```csharp
// ── 空间离散化 ──
// 连续 ℝ³ ↔ 离散 ℤ³ 的映射

public readonly struct VoxelIndex
{
    public readonly int X, Y, Z;
    
    public VoxelIndex(int x, int y, int z) { X = x; Y = y; Z = z; }
    
    // 从物理坐标计算体素索引
    public static VoxelIndex FromWorld(Vector3 worldPos, float resolution)
    {
        return new VoxelIndex(
            (int)MathF.Floor(worldPos.X / resolution),
            (int)MathF.Floor(worldPos.Y / resolution),
            (int)MathF.Floor(worldPos.Z / resolution)
        );
    }
    
    // 从体素索引计算体素中心的世界坐标
    public Vector3 ToWorldCenter(float resolution)
    {
        return new Vector3(
            (X + 0.5f) * resolution,
            (Y + 0.5f) * resolution,
            (Z + 0.5f) * resolution
        );
    }
    
    // 体素边界（AABB）
    public Aabb ToWorldAabb(float resolution)
    {
        var min = new Vector3(X * resolution, Y * resolution, Z * resolution);
        var max = new Vector3((X + 1) * resolution, (Y + 1) * resolution, (Z + 1) * resolution);
        return new Aabb(min, max - min);
    }
}

// ── 空间离散化骨架 ──
// 将物理坐标转换为体素索引

public class SpatialDiscretizationSkeleton : SkeletonBase<Vector3, VoxelIndex>
{
    private readonly float _resolution;
    
    public SpatialDiscretizationSkeleton(float resolution = 0.1f)
        : base("SpatialDiscretization", $"空间离散化：ℝ³ → ℤ³ (R={resolution}m)")
    {
        _resolution = resolution;
    }
    
    protected override VoxelIndex ExecuteCore(Vector3 worldPos, SkeletonContext ctx)
    {
        return VoxelIndex.FromWorld(worldPos, _resolution);
    }
}
```

## 3. 分辨率函数与多分辨率共存

### 3.1 数学定义

```
分辨率函数 R: ℝ³ → ℝ⁺ 是空间位置的函数：

  R(x) = 分辨率在位置 x 的值

非紧耦合世界的核心：分辨率可以空间变化

  R(x) = 
    R_min          当 ||x - x_player|| < d_near（玩家附近，高精度）
    R_min · 2^k    当 d_near · 2^{k-1} ≤ ||x - x_player|| < d_near · 2^k（LOD 区域）
    ∞              当超出世界边界（未加载区域）
```

### 3.2 多分辨率体素

```
不同分辨率的体素在同一个世界中共存：

  高分辨率区域 (R=1cm)         低分辨率区域 (R=10cm)
  ┌──┬──┬──┬──┬──┐            ┌──────┬──────┐
  │  │  │  │  │  │            │      │      │
  ├──┼──┼──┼──┼──┤            │ 1cm³ │ 1cm³ │
  │  │  │  │  │  │            │ 体素  │ 体素  │
  ├──┼──┼──┼──┼──┤            │      │      │
  │  │  │  │  │  │            ├──────┼──────┤
  ├──┼──┼──┼──┼──┤            │      │      │
  │  │  │  │  │  │            │ 1cm³ │ 1cm³ │
  └──┴──┴──┴──┴──┘            │ 体素  │ 体素  │
                               └──────┴──────┘
  分辨率 R=0.01m               分辨率 R=0.01m（相同）

  不同分辨率区域的边界：
  ┌──────┬──┬──┬──┐
  │      │  │  │  │
  │ R=10 │  │ R=1 │
  │ cm   │  │ cm  │
  │      ├──┼──┼──┤
  │      │  │  │  │
  └──────┴──┴──┴──┘
  
  边界处：低分辨率体素映射到多个高分辨率体素
  10cm³ 体素 = 10×10×10 个 1cm³ 体素
```

### 3.3 编程实现

```csharp
// ── 分辨率函数 ──
// 根据玩家位置计算空间变化的分辨率

public class ResolutionFunction
{
    private readonly float _baseResolution;  // 基础分辨率（最近处）
    private readonly float _nearDistance;     // 高精度区域半径
    private readonly int _lodLevels;          // LOD 级数
    
    public ResolutionFunction(float baseResolution = 0.01f, float nearDistance = 5f, int lodLevels = 3)
    {
        _baseResolution = baseResolution;
        _nearDistance = nearDistance;
        _lodLevels = lodLevels;
    }
    
    // 计算位置 x 处的分辨率
    public float GetResolution(Vector3 position, Vector3 playerPosition)
    {
        float dist = (position - playerPosition).Length();
        
        if (dist < _nearDistance)
            return _baseResolution;
        
        // LOD 区域：每翻一倍距离，分辨率翻倍
        float lodDist = _nearDistance;
        for (int i = 1; i <= _lodLevels; i++)
        {
            lodDist *= 2f;
            if (dist < lodDist)
                return _baseResolution * (1 << i);
        }
        
        // 超出 LOD 范围，返回最大分辨率（或标记为未加载）
        return _baseResolution * (1 << _lodLevels);
    }
    
    // 获取位置 x 处的体素大小（体素边长，米）
    public float GetVoxelSize(Vector3 position, Vector3 playerPosition)
    {
        return GetResolution(position, playerPosition);
    }
    
    // 判断两个不同分辨率的体素是否相邻
    public static bool AreAdjacent(VoxelIndex a, float resA, VoxelIndex b, float resB)
    {
        // 计算两个体素的 AABB
        var aabbA = a.ToWorldAabb(resA);
        var aabbB = b.ToWorldAabb(resB);
        
        // 检查是否相邻（共享面）
        return aabbA.Intersects(aabbB.Grow(0.001f));
    }
}

// ── 多分辨率体素骨架 ──
// 根据玩家位置动态选择分辨率

public class AdaptiveResolutionSkeleton : SkeletonBase<Vector3, (VoxelIndex, float)>
{
    private readonly ResolutionFunction _resolutionFunc;
    private Vector3 _playerPosition;
    
    public AdaptiveResolutionSkeleton(ResolutionFunction resolutionFunc)
        : base("AdaptiveResolution", "自适应分辨率：根据距离选择体素大小")
    {
        _resolutionFunc = resolutionFunc;
    }
    
    public void UpdatePlayerPosition(Vector3 playerPos)
    {
        _playerPosition = playerPos;
    }
    
    protected override (VoxelIndex, float) ExecuteCore(Vector3 worldPos, SkeletonContext ctx)
    {
        float resolution = _resolutionFunc.GetResolution(worldPos, _playerPosition);
        var index = VoxelIndex.FromWorld(worldPos, resolution);
        return (index, resolution);
    }
}
```

## 4. 体素块组织

### 4.1 层次结构

```
体素世界 = 层次化的块组织

  体素 (i,j,k) ∈ ℤ³                    ← 最小单位
     ↓ 分组
  SubChunk (8×8×8 体素)                 ← 局部缓存单位
     ↓ 分组
  Chunk (32×64×32 体素)                 ← 加载/卸载/网格构建单位
     ↓ 分组
  Region (8×8×8 个 Chunk)               ← 存储/序列化单位
     ↓
  世界管理器                              ← 全局管理
```

### 4.2 数学定义

```
Chunk 坐标到体素坐标：
  Chunk (cx, cy, cz) 包含体素：
    i ∈ [cx · SizeX, (cx+1) · SizeX)
    j ∈ [cy · SizeY, (cy+1) · SizeY)
    k ∈ [cz · SizeZ, (cz+1) · SizeZ)

体素坐标到 Chunk 坐标：
  cx = floor(i / SizeX)
  cy = floor(j / SizeY)
  cz = floor(k / SizeZ)

Region 坐标到 Chunk 坐标：
  Region (rx, ry, rz) 包含 Chunk：
    cx ∈ [rx · ChunksPerDim, (rx+1) · ChunksPerDim)
    cy ∈ [ry · ChunksPerDim, (ry+1) · ChunksPerDim)
    cz ∈ [rz · ChunksPerDim, (rz+1) · ChunksPerDim)
```

### 4.3 编程实现

```csharp
// ── 体素块组织 ──
// Chunk/Region 层次结构

public static class ChunkMath
{
    // 体素坐标 → Chunk 坐标
    public static int ToChunkCoord(int voxelCoord, int chunkSize)
    {
        return voxelCoord >= 0 
            ? voxelCoord / chunkSize 
            : (voxelCoord - chunkSize + 1) / chunkSize;
    }
    
    // 体素坐标 → Chunk 内局部坐标
    public static int ToLocalCoord(int voxelCoord, int chunkSize)
    {
        int c = ToChunkCoord(voxelCoord, chunkSize);
        return voxelCoord - c * chunkSize;
    }
    
    // Chunk 坐标 → Region 坐标
    public static int ToRegionCoord(int chunkCoord, int chunksPerDim)
    {
        return chunkCoord >= 0
            ? chunkCoord / chunksPerDim
            : (chunkCoord - chunksPerDim + 1) / chunksPerDim;
    }
}

// ── 体素块组织骨架 ──
// 将体素索引映射到 Chunk/Region

public class ChunkOrganizationSkeleton : SkeletonBase<VoxelIndex, (int, int, int)>
{
    private readonly int _chunkSizeX, _chunkSizeY, _chunkSizeZ;
    
    public ChunkOrganizationSkeleton(
        int chunkSizeX = 32, int chunkSizeY = 64, int chunkSizeZ = 32)
        : base("ChunkOrganization", "体素 → Chunk 坐标映射")
    {
        _chunkSizeX = chunkSizeX;
        _chunkSizeY = chunkSizeY;
        _chunkSizeZ = chunkSizeZ;
    }
    
    protected override (int, int, int) ExecuteCore(VoxelIndex voxel, SkeletonContext ctx)
    {
        return (
            ChunkMath.ToChunkCoord(voxel.X, _chunkSizeX),
            ChunkMath.ToChunkCoord(voxel.Y, _chunkSizeY),
            ChunkMath.ToChunkCoord(voxel.Z, _chunkSizeZ)
        );
    }
}
```

## 5. 非紧耦合世界

### 5.1 什么是非紧耦合

```
紧耦合世界：所有体素通过连续网格紧密连接
  ℱ_micro(x) 和 ℱ_micro(x+δx) 通过 ∇² 算子直接耦合
  每个体素的演化依赖其邻居

非紧耦合世界：区域之间通过接口松散连接
  区域 A ──边界接口── 区域 B
     │                    │
     │ 内部：密集网格       │ 内部：稀疏网格
     │ 演化：每帧更新       │ 演化：每10帧更新
     │ 分辨率：1cm³         │ 分辨率：10cm³
     │ 物理：固体           │ 物理：流体
```

### 5.2 非紧耦合的数学形式

```
区域内部（紧耦合）：
  ∂ℱ_micro/∂t = D · ∇²ℱ_micro + S[ℱ_micro, ε]    ← 标准 PDE

区域边界（松耦合）：
  ℱ_micro_A(boundary) = T_AB(ℱ_micro_B(boundary))  ← 接口变换

其中 T_AB 是区域 A 和 B 之间的接口变换函数：
  T_AB: 𝕄_B → 𝕄_A
  T_AB 处理分辨率差异、时间步长差异、物理差异
```

### 5.3 接口变换

```
不同分辨率区域之间的接口：

  高分辨率区域 (R=1cm)          低分辨率区域 (R=10cm)
  ┌──┬──┬──┬──┬──┐
  │  │  │  │  │  │
  ├──┼──┼──┼──┼──┤     T_AB     ┌──────────┐
  │  │  │  │  │  │  ────────→   │          │
  ├──┼──┼──┼──┼──┤    聚合      │ 10cm³    │
  │  │  │  │  │  │  ←────────   │ 体素     │
  ├──┼──┼──┼──┼──┤    插值      │          │
  │  │  │  │  │  │              └──────────┘
  └──┴──┴──┴──┴──┘

  聚合：10×10×10 个 1cm³ 体素 → 1 个 10cm³ 体素
    ℱ_micro_coarse = (1/N) · Σ ℱ_micro_fine(i)   （平均）
    
  插值：1 个 10cm³ 体素 → 10×10×10 个 1cm³ 体素
    ℱ_micro_fine(i) = ℱ_micro_coarse + ∇ℱ_micro · δx_i  （线性插值）
```

### 5.4 编程实现

```csharp
// ── 区域接口 ──
// 不同分辨率/不同区域之间的连接

public class RegionInterface
{
    public string Name { get; }
    public float ResolutionA { get; }  // 区域 A 的分辨率
    public float ResolutionB { get; }  // 区域 B 的分辨率
    
    public RegionInterface(string name, float resA, float resB)
    {
        Name = name;
        ResolutionA = resA;
        ResolutionB = resB;
    }
    
    // 从高分辨率到低分辨率（聚合）
    public MicroState Aggregate(IEnumerable<MicroState> fineStates)
    {
        int count = 0;
        float m = 0, b = 0, f = 0, v = 0;
        foreach (var state in fineStates)
        {
            m += state.M; b += state.B; f += state.F; v += state.V;
            count++;
        }
        if (count == 0) return new MicroState();
        return new MicroState
        {
            M = (int)(m / count),
            B = (int)(b / count),
            F = (int)(f / count),
            V = (int)(v / count),
        };
    }
    
    // 从低分辨率到高分辨率（插值）
    public MicroState[] Interpolate(MicroState coarse, int factor)
    {
        int total = factor * factor * factor;
        var result = new MicroState[total];
        for (int i = 0; i < total; i++)
            result[i] = coarse;  // 简化：零阶保持
        return result;
    }
}

// ── 非紧耦合区域 ──
// 一个松散连接的世界区域

public class WorldRegion
{
    public string Id { get; }
    public Aabb Bounds { get; }           // 区域边界
    public float Resolution { get; }      // 区域分辨率
    public float TimeScale { get; }       // 时间缩放（每 N 帧更新一次）
    public Dictionary<string, RegionInterface> Interfaces { get; } = new();
    
    public WorldRegion(string id, Aabb bounds, float resolution, float timeScale = 1.0f)
    {
        Id = id;
        Bounds = bounds;
        Resolution = resolution;
        TimeScale = timeScale;
    }
    
    // 添加与其他区域的接口
    public void AddInterface(string otherRegionId, RegionInterface iface)
    {
        Interfaces[otherRegionId] = iface;
    }
    
    // 判断体素是否属于此区域
    public bool Contains(VoxelIndex voxel)
    {
        var worldPos = voxel.ToWorldCenter(Resolution);
        return Bounds.Contains(worldPos);
    }
}
```

## 6. 世界生成管线

### 6.1 从硬编码到骨架驱动

```
传统方式（硬编码）：
  GenerateTerrain(ref data, seed, frequency, octaves, ...) {
      for each (x, z) {
          height = SimpleNoise(...)
          for each y {
              if (y == 0) blocks[index] = Bedrock
              else if (y < height-4) blocks[index] = Granite
              ...
          }
      }
  }

骨架驱动方式：
  世界生成 = Flow.Start<WorldConfig>()
      .Then(地形骨架)        // 用 ℱ_micro 表达地形
      .Then(生态骨架)        // 用 ε 参数表达生态
      .Then(结构骨架)        // 用联系矩阵放置结构
      .Then(区块化骨架)      // 分块存储
```

### 6.2 世界生成管线架构

```
WorldConfig (种子、分辨率、世界大小)
    │
    ▼
地形生成骨架 ───→ ℱ_micro(x) = (M,B,F,V)  ← 每个体素的微观状态
    │
    ▼
生态分配骨架 ───→ ε(x) = (湿度, 温度, 风化度, ...)  ← 每个体素的 ε 参数
    │
    ▼
结构放置骨架 ───→ 修改特定位置的 ℱ_micro 和 ε  ← 树木、建筑、洞穴
    │
    ▼
区块化骨架 ─────→ Chunk[][][]  ← 分块存储到 HighResChunk
    │
    ▼
宏观映射骨架 ───→ ℱ_macro(x)  ← 可选：预计算宏观属性
```

## 7. 地形生成骨架

### 7.1 数学

```
地形 ℱ_micro(x) 由高度场 h(x,z) 决定：

  h(x,z) = Σ_{k=1}^{octaves} gain^{k-1} · noise(frequency · lacunarity^{k-1} · x, z)

  对于体素 (i,j,k)：
    如果 j < h(i,k) - 4：  ℱ_micro = (M=高, B=高, F=低, V=低)  ← 基岩/石头
    如果 j < h(i,k) - 1：  ℱ_micro = (M=中, B=中, F=中, V=低)  ← 泥土
    如果 j = h(i,k) - 1：  ℱ_micro = (M=中, B=高, F=低, V=低)  ← 草方块
    如果 j > h(i,k)：      ℱ_micro = (0,0,0,0)                  ← 空气
```

### 7.2 编程实现

```csharp
// ── 地形生成骨架 ──
// 用 ℱ_micro 表达地形

public class TerrainGenerationSkeleton : SkeletonBase<WorldGenContext, WorldGenContext>
{
    public TerrainGenerationSkeleton()
        : base("TerrainGeneration", "地形生成：用 ℱ_micro 表达地形") { }
    
    protected override WorldGenContext ExecuteCore(WorldGenContext ctx, SkeletonContext skelCtx)
    {
        var config = ctx.Config;
        var noise = new NoiseGenerator(config.Seed);
        
        // 遍历世界中的所有体素位置
        // 注意：实际实现中，只生成 Chunk 范围内的体素
        for (int x = 0; x < 32; x++)      // Chunk SizeX
        {
            for (int z = 0; z < 32; z++)  // Chunk SizeZ
            {
                // 计算高度
                float heightVal = noise.OctaveNoise(
                    x, z, config.Frequency, config.Octaves, 
                    config.Lacunarity, config.Gain);
                int height = (int)((heightVal + 1f) * 15f) + 30;
                height = Math.Max(4, Math.Min(config.WorldHeight - 1, height));
                
                for (int y = 0; y < config.WorldHeight; y++)
                {
                    var index = new VoxelIndex(x, y, z);
                    
                    MicroState micro;
                    if (y == 0)
                        micro = new MicroState { M = 200, B = 150, F = 10, V = 5 };   // 基岩
                    else if (y < height - 4)
                        micro = new MicroState { M = 150, B = 120, F = 20, V = 10 };  // 石头
                    else if (y < height - 1)
                        micro = new MicroState { M = 80, B = 60, F = 40, V = 20 };    // 泥土
                    else if (y == height - 1)
                        micro = new MicroState { M = 60, B = 100, F = 15, V = 5 };    // 草方块
                    else
                        micro = new MicroState { M = 0, B = 0, F = 0, V = 0 };        // 空气
                    
                    ctx.MicroStates[index] = micro;
                }
            }
        }
        
        return ctx;
    }
}

// ── 噪声生成器 ──
// 简化版 Perlin 噪声

public class NoiseGenerator
{
    private readonly int _seed;
    
    public NoiseGenerator(int seed) { _seed = seed; }
    
    public float OctaveNoise(float x, float z, float frequency, int octaves, 
        float lacunarity, float gain)
    {
        float value = 0;
        float amplitude = 1;
        float maxValue = 0;
        float freq = frequency;
        
        for (int i = 0; i < octaves; i++)
        {
            int n = (int)(x * freq * 374761393 + z * freq * 668265263 + 
                         (_seed + i * 1000) * 2246822519);
            n = (n ^ (n >> 13)) * 1274126177;
            n = n ^ (n >> 16);
            float noise = (2.0f * ((n & 0x7FFFFFFF) / (float)0x7FFFFFFF)) - 1.0f;
            value += noise * amplitude;
            maxValue += amplitude;
            amplitude *= gain;
            freq *= lacunarity;
        }
        
        return value / maxValue;
    }
}
```

### 9.2 编程实现

```csharp
// ── 区块化骨架 ──
// 将体素数据分块存储到 HighResChunk

public class ChunkingSkeleton : SkeletonBase<WorldGenContext, HighResChunk[]>
{
    private readonly int _chunkSizeX = 32;
    private readonly int _chunkSizeY = 64;
    private readonly int _chunkSizeZ = 32;
    
    public ChunkingSkeleton()
        : base("Chunking", "区块化：将体素数据分块存储") { }
    
    protected override HighResChunk[] ExecuteCore(WorldGenContext ctx, SkeletonContext skelCtx)
    {
        // 按 Chunk 坐标分组
        var chunkGroups = new Dictionary<(int, int, int), List<(VoxelIndex, MicroState)>>();
        
        foreach (var (index, micro) in ctx.MicroStates)
        {
            int cx = ChunkMath.ToChunkCoord(index.X, _chunkSizeX);
            int cy = ChunkMath.ToChunkCoord(index.Y, _chunkSizeY);
            int cz = ChunkMath.ToChunkCoord(index.Z, _chunkSizeZ);
            var key = (cx, cy, cz);
            
            if (!chunkGroups.ContainsKey(key))
                chunkGroups[key] = new List<(VoxelIndex, MicroState)>();
            chunkGroups[key].Add((index, micro));
        }
        
        var chunks = new List<HighResChunk>();
        foreach (var ((cx, cy, cz), voxels) in chunkGroups)
        {
            var chunk = new HighResChunk(cx, cy, cz);
            
            foreach (var (index, micro) in voxels)
            {
                int lx = ChunkMath.ToLocalCoord(index.X, _chunkSizeX);
                int ly = ChunkMath.ToLocalCoord(index.Y, _chunkSizeY);
                int lz = ChunkMath.ToLocalCoord(index.Z, _chunkSizeZ);
                
                // 将 MicroState 转换为 Voxel
                var voxel = MicroStateToVoxel(micro);
                chunk.SetVoxel(lx, ly, lz, voxel);
            }
            
            chunk.NeedsLODUpdate = true;
            chunk.NeedsMeshBuild = true;
            chunks.Add(chunk);
        }
        
        return chunks.ToArray();
    }
    
    // MicroState → Voxel 转换
    private static Voxel MicroStateToVoxel(MicroState micro)
    {
        if (micro.IsEmpty)
            return Voxel.Empty;
        
        // 根据 M:B:F:V 比例选择方块类型
        // 这是一个简化映射，实际应根据具体游戏逻辑定义
        BlockID block;
        if (micro.M > 150) block = BlockID.Stone;
        else if (micro.B > 80) block = BlockID.Grass;
        else if (micro.M > 50) block = BlockID.Dirt;
        else block = BlockID.Air;
        
        return new Voxel(block, MaterialID.Default);
    }
}
```

## 11. LOD 系统

### 11.1 数学

```
LOD（Level of Detail）= 多分辨率体素表示

LOD 级别：
  LOD 0：全分辨率（R = R_base）
  LOD 1：半分辨率（R = 2 · R_base），8 个体素合并为 1 个
  LOD 2：四分之一分辨率（R = 4 · R_base），64 个体素合并为 1 个

LOD 选择：
  LOD_level(x) = floor(log₂(||x - x_player|| / d_near))
```

### 11.2 编程实现

```csharp
// ── LOD 系统 ──
// 多分辨率体素表示

public class LodSystem
{
    private readonly float _baseResolution;
    private readonly float _nearDistance;
    
    public LodSystem(float baseResolution = 0.01f, float nearDistance = 5f)
    {
        _baseResolution = baseResolution;
        _nearDistance = nearDistance;
    }
    
    // 计算位置 x 处的 LOD 级别
    public int GetLodLevel(Vector3 position, Vector3 playerPosition)
    {
        float dist = (position - playerPosition).Length();
        if (dist < _nearDistance) return 0;
        
        int level = (int)MathF.Floor(MathF.Log(dist / _nearDistance) / MathF.Log(2));
        return Math.Clamp(level, 0, 3);
    }
    
    // 从全分辨率体素构建 LOD
    public Voxel BuildLodVoxel(HighResChunk chunk, int lodLevel, int lx, int ly, int lz)
    {
        if (lodLevel == 0)
            return chunk.GetVoxel(lx, ly, lz);
        
        int step = 1 << lodLevel;  // 2^lodLevel
        int halfStep = step / 2;
        
        // 合并 step×step×step 个体素
        int solidCount = 0;
        BlockID dominantBlock = BlockID.Air;
        var dominantCounts = new Dictionary<BlockID, int>();
        
        for (int dx = 0; dx < step; dx++)
        for (int dy = 0; dy < step; dy++)
        for (int dz = 0; dz < step; dz++)
        {
            int sx = lx * step + dx;
            int sy = ly * step + dy;
            int sz = lz * step + dz;
            
            if (sx >= HighResChunk.SizeX || sy >= HighResChunk.SizeY || sz >= HighResChunk.SizeZ)
                continue;
            
            var voxel = chunk.GetVoxel(sx, sy, sz);
            if (voxel.Block.IsSolid)
            {
                solidCount++;
                if (!dominantCounts.ContainsKey(voxel.Block))
                    dominantCounts[voxel.Block] = 0;
                dominantCounts[voxel.Block]++;
            }
        }
        
        // 选择出现最多的方块类型
        int maxCount = 0;
        foreach (var (block, count) in dominantCounts)
        {
            if (count > maxCount)
            {
                maxCount = count;
                dominantBlock = block;
            }
        }
        
        // 如果超过一半是固体，则 LOD 体素为固体
        int total = step * step * step;
        return solidCount > total / 2 
            ? new Voxel(dominantBlock, MaterialID.Default) 
            : Voxel.Empty;
    }
}

// ── LOD 构建骨架 ──
// 为 Chunk 构建 LOD 缓存

public class LodBuildSkeleton : SkeletonBase<VoxelGameContext, VoxelGameContext>
{
    public LodBuildSkeleton()
        : base("LodBuild", "LOD 构建：为 Chunk 构建多分辨率缓存") { }
    
    protected override VoxelGameContext ExecuteCore(VoxelGameContext ctx, SkeletonContext skelCtx)
    {
        var sw = System.Diagnostics.Stopwatch.StartNew();
        float budgetMs = ctx.Budget?.Allocate(4.0f) ?? 4.0f;
        int built = 0;
        
        foreach (var chunk in ctx.WorldManager.AllActiveChunks)
        {
            if (sw.ElapsedMilliseconds >= budgetMs) break;
            if (!chunk.NeedsLODUpdate) continue;
            
            // 构建 LOD 1 和 LOD 2
            chunk.LODs[1] = LodVolume.FromChunk(chunk, 1);
            chunk.LODs[2] = LodVolume.FromChunk(chunk, 2);
            chunk.NeedsLODUpdate = false;
            built++;
        }
        
        ctx.FrameStats["LodBuiltCount"] = built;
        ctx.FrameStats["LodBuildMs"] = (int)sw.ElapsedMilliseconds;
        
        return ctx;
    }
}
```

```
体素索引 (i, j, k) ∈ ℤ³    ← 分辨率 R=0.01m → 1cm³
    │
    ▼ 块组织
Chunk 坐标 (cx, cy, cz)     ← Chunk 大小 32×64×32 体素
    │                          → 物理大小 32cm × 64cm × 32cm
    ▼ 世界生成
ℱ_micro(i,j,k) = (M,B,F,V)  ← 每个 1cm³ 体素的微观状态
    │
    ▼ 宏观映射
ℱ_macro(i,j,k) = lim_{ε→0} 𝒜_ε[ℱ_micro]  ← 每个 1cm³ 体素的宏观属性
    │
    ▼ 网格构建
Mesh (顶点, 法线, UV)       ← 每个 1cm³ 体素的面
    │
    ▼ 渲染
屏幕上的方块                ← 1cm³ 体素 = 1 个方块
```
