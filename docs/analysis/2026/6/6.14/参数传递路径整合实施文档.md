#### 问题 A：三套平行的接口体系

| 命名空间 | 接口 | 实现类 | 行数 |

| `Share.Registry` | `IFunctionExecutor` | `CommandFramework.Execution.FunctionExecutor` | 246 行 |

| 位置 | 方法 | 输入 | 策略 |
|------|------|------|------|
| `CommandFramework.Execution.FunctionExecution.cs:108` | `BuildArgumentsWithResolver()` | `string?[]` / `object?[]` / 其他 | 4 条快速路径 + ParameterBuilder 降级 |

| 6 | 延迟初始化 | `InitializeDefault()` / `Initialize()` |

#### 问题 D：非事件命令注册路径割裂

```
注册一条非事件命令需要 4 步联动：
  UnifiedCommandRegistry.Register()
```

## 4. Step 1：统一参数构建逻辑

### 4.1 现状

两个独立的参数构建器：

```
CommandFramework.Execution.FunctionExecution.cs
  └─ BuildArgumentsWithResolver(IFunctionDescriptor, object?)
      ├─ 快速路径 1：object?[] 长度匹配 → 类型检查后直接返回
      ├─ 快速路径 2：string?[] 长度匹配 → TypeConverter.ConvertArguments
      ├─ 快速路径 3：object?[] 长度不匹配 → 截取/补全
      └─ 降级路径：ParameterBuilder.Build()

FunctionStorage.Core.Utilities.cs
  └─ ArgumentBuilder.BuildArguments(Delegate, object?)
      ├─ 情况 1：IReadOnlyDictionary → 按参数名匹配
      ├─ 情况 2：object?[] → 位置参数 + 类型转换
      └─ 情况 3：其他 → 包装为单元素数组
```

### 4.2 目标

将两套逻辑合并为公共类 `ArgumentBuilder`，放在 `Share/CommandFramework/Execution/` 下，两处执行器统一使用。

**关键约束**：`Share` 项目不能引用 `Operation` 项目，因此 `ArgumentBuilder` 中不能出现 `FunctionStorage.Core.TypeConverter` 的引用。所有类型转换统一使用 `ParameterResolver.Converter.TypeConverter`。

### 4.3 改动详情

#### 文件 1：新建 `Share/CommandFramework/Execution/ArgumentBuilder.cs`

```csharp
// ============================================================
// ArgumentBuilder.cs — 统一参数构建器
//
// 合并自：
//   CommandFramework.Execution.FunctionExecution.BuildArgumentsWithResolver
//   FunctionStorage.Core.ArgumentBuilder.BuildArguments
//
// 提供统一的参数构建入口，支持：
//   1. string?[] → TypeConverter.ConvertArguments（快速路径）
//   2. object?[] → 类型检查 + 转换
//   3. IReadOnlyDictionary<string, object?> → 按参数名匹配
//   4. 其他 → ParameterBuilder 降级
//
// ★ 注意：所有类型转换统一使用 ParameterResolver.Converter.TypeConverter，
//   不引用 FunctionStorage.Core.TypeConverter（避免循环依赖）。
// ============================================================

using System.Reflection;
using ParameterResolver.Abstractions;
using ParameterResolver.Builder;
using ParameterResolver.Converter;
using Share.Registry;

namespace CommandFramework.Execution;

public static class ArgumentBuilder
{
    // ─── 公开入口 ───

    public static object?[]? BuildFromDescriptor(
        IFunctionDescriptor descriptor, object? parameters)
    {
        if (descriptor.Parameters.Count == 0)
            return Array.Empty<object>();

        var paramTypes = descriptor.Parameters.Select(p => p.Type).ToArray();

        if (parameters is string?[] stringArgs && stringArgs.Length == paramTypes.Length)
            return ConvertStringArray(stringArgs, paramTypes);

        if (parameters is object?[] objArgs)
            return ConvertObjectArray(objArgs, paramTypes);

        return BuildViaParameterBuilder(descriptor, parameters);
    }

    public static object?[] BuildFromDelegate(
        Delegate del, object? parameters)
    {
        var methodInfo = del.Method;
        var paramInfos = methodInfo.GetParameters();

        if (paramInfos.Length == 0)
            return Array.Empty<object?>();

        if (parameters is IReadOnlyDictionary<string, object?> dict)
            return BuildFromDictionary(paramInfos, dict);

        if (parameters is IReadOnlyDictionary<string, string?> stringDict)
        {
            var objDict = new Dictionary<string, object?>(StringComparer.OrdinalIgnoreCase);
            foreach (var kvp in stringDict)
                objDict[kvp.Key] = kvp.Value;
            return BuildFromDictionary(paramInfos, objDict);
        }

        if (parameters is object?[] arr)
            return ConvertObjectArrayToParamInfos(arr, paramInfos);

        return new[] { parameters };
    }

    // ─── 私有方法 ───

    private static object?[] ConvertStringArray(string?[] stringArgs, Type[] paramTypes)
    {
        var converted = TypeConverter.ConvertArguments(stringArgs!, paramTypes);
        for (int i = 0; i < converted.Length; i++)
        {
            if (converted[i] is string s && paramTypes[i] != typeof(string))
            {
                try
                {
                    converted[i] = System.Convert.ChangeType(
                        s, paramTypes[i], System.Globalization.CultureInfo.InvariantCulture);
                }
                catch { }
            }
        }
        return converted;
    }

    private static object?[] ConvertObjectArray(object?[] objArgs, Type[] paramTypes)
    {
        if (objArgs.Length == paramTypes.Length)
        {
            bool needsConversion = false;
            for (int i = 0; i < objArgs.Length; i++)
            {
                if (objArgs[i] != null && !paramTypes[i].IsInstanceOfType(objArgs[i]))
                { needsConversion = true; break; }
            }
            if (!needsConversion) return objArgs;

            var converted = new object?[objArgs.Length];
            for (int i = 0; i < objArgs.Length; i++)
            {
                var (cv, _) = TypeConverter.Convert(objArgs[i], paramTypes[i]);
                converted[i] = cv ?? objArgs[i];
            }
            return converted;
        }

        var minLen = Math.Min(objArgs.Length, paramTypes.Length);
        var result = new object?[paramTypes.Length];
        for (int i = 0; i < paramTypes.Length; i++)
        {
            if (i < minLen)
            {
                var (cv, _) = TypeConverter.Convert(objArgs[i], paramTypes[i]);
                result[i] = cv ?? objArgs[i];
            }
        }
        return result;
    }

    private static object?[]? BuildViaParameterBuilder(
        IFunctionDescriptor descriptor, object? parameters)
    {
        var paramDescriptors = descriptor.Parameters
            .Select(p => new ParameterDescriptorAdapter(p))
            .ToList();
        var builder = new ParameterBuilder();
        var result = builder.Build(paramDescriptors, parameters);
        return result.IsSuccess ? result.Args : null;
    }

    private static object?[] BuildFromDictionary(
        ParameterInfo[] paramInfos,
        IReadOnlyDictionary<string, object?> dict)
    {
        var result = new object?[paramInfos.Length];
        var matched = new bool[paramInfos.Length];

        for (int i = 0; i < paramInfos.Length; i++)
        {
            var paramName = paramInfos[i].Name ?? $"param{i}";
            if (dict.TryGetValue(paramName, out var value))
            {
                var (cv, _) = TypeConverter.Convert(value, paramInfos[i].ParameterType);
                result[i] = cv ?? value;
                matched[i] = true;
            }
        }

        for (int i = 0; i < paramInfos.Length; i++)
        {
            if (!matched[i])
            {
                if (paramInfos[i].HasDefaultValue)
                    result[i] = paramInfos[i].DefaultValue;
                else if (paramInfos[i].ParameterType.IsValueType)
                    result[i] = Activator.CreateInstance(paramInfos[i].ParameterType);
                else
                    result[i] = null;
            }
        }
        return result;
    }

    private static object?[] ConvertObjectArrayToParamInfos(
        object?[] arr, ParameterInfo[] paramInfos)
    {
        var result = new object?[paramInfos.Length];
        for (int i = 0; i < paramInfos.Length && i < arr.Length; i++)
        {
            var (cv, _) = TypeConverter.Convert(arr[i], paramInfos[i].ParameterType);
            result[i] = cv ?? arr[i];
        }
        for (int i = arr.Length; i < paramInfos.Length; i++)
        {
            if (paramInfos[i].HasDefaultValue)
                result[i] = paramInfos[i].DefaultValue;
            else if (paramInfos[i].ParameterType.IsValueType)
                result[i] = Activator.CreateInstance(paramInfos[i].ParameterType);
            else
                result[i] = null;
        }
        return result;
    }
}
```

```csharp
// FunctionStorageRegistry.cs — 统一函数注册入口（精简门面）
//
// 职责：仅提供静态注册/执行门面，委托给底层实现。
// 非事件命令管理 → NonEventCommandCenter
// IFunctionUnit 存储 → FunctionStorageAdapter
// ============================================================

using Share.Registry;
using CommandFramework.Execution;

namespace FunctionStorage;

public static class FunctionStorageRegistry
{
    private static Share.Registry.FunctionRegistry? _registry;
    private static CommandFramework.Execution.FunctionExecutor? _executor;
    private static readonly object _lock = new();

    public static Share.Registry.FunctionRegistry Registry
    {
        get
        {
            if (_registry == null) InitializeDefault();
            return _registry!;
        }
    }

    public static CommandFramework.Execution.FunctionExecutor Executor
    {
        get
        {
            if (_executor == null) InitializeDefault();
            return _executor!;
        }
    }

    public static void Initialize(
        Share.Registry.FunctionRegistry registry,
        CommandFramework.Execution.FunctionExecutor executor)
    {
        ArgumentNullException.ThrowIfNull(registry);
        ArgumentNullException.ThrowIfNull(executor);
        lock (_lock)
        {
            _registry = registry;
            _executor = executor;
        }
    }

    private static void InitializeDefault()
    {
        lock (_lock)
        {
            _registry ??= new Share.Registry.FunctionRegistry();
            _executor ??= new CommandFramework.Execution.FunctionExecutor(_registry);
        }
    }

    // ─── 注册门面 ───

    public static void Register(string name, Delegate function, string? description = null)
        => Registry.Register(name, function, description);

    public static void Unregister(string name)
        => Registry.Unregister(name);

    public static bool Contains(string name)
        => Registry.Contains(name);

    public static IEnumerable<string> GetRegisteredNames()
        => Registry.GetRegisteredNames();

    public static bool TryGetDescriptor(string name, out IFunctionDescriptor descriptor)
        => Registry.TryGetDescriptor(name, out descriptor);

    // ─── 执行门面 ───

    public static async Task<IFunctionExecutionResult> ExecuteAsync(
        string functionName, object? parameters = null)
        => await Executor.ExecuteAsync(functionName, parameters);

    public static async Task<IFunctionExecutionResult> ExecuteAsync(
        Delegate function, object? parameters = null)
        => await Executor.ExecuteAsync(function, parameters);

    public static void Clear()
    {
        Registry.Clear();
        NonEventCommandCenter.Clear();
        FunctionStorageAdapter.Clear();
    }
}
```

```csharp
    }
```

```csharp
// ============================================================
// FunctionStorageAdapter.cs — 函数存储适配器
//
// 从 FunctionStorageRegistry 拆分出的 IFunctionUnit 存储职责。
// ============================================================

using System.Collections.Concurrent;
using FunctionStorage.Abstractions;

namespace FunctionStorage;

public static class FunctionStorageAdapter
{
    private static ConcurrentDictionary<string, IFunctionUnit>? _units;

    public static void RegisterUnit(IFunctionUnit unit)
    {
        ArgumentNullException.ThrowIfNull(unit);
        if (string.IsNullOrWhiteSpace(unit.Name))
            throw new ArgumentException("Unit name cannot be empty", nameof(unit));

        FunctionStorageRegistry.Register(unit.Name,
            (Func<object, Task<object>>)(async (input) =>
            {
                var result = await unit.ExecuteAsync(input);
                return result.Value!;
            }), unit.Description);

        _units ??= new ConcurrentDictionary<string, IFunctionUnit>(
            StringComparer.OrdinalIgnoreCase);
        _units[unit.Name] = unit;
    }

    public static bool TryGetUnit(string name, out IFunctionUnit? unit)
    {
        if (_units == null)
        {
            unit = null;
            return false;
        }
        return _units.TryGetValue(name, out unit);
    }

    public static void Clear()
    {
        _units?.Clear();
    }
}
```
