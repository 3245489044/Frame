# 通用注册表框架（Universal Registry Framework）

## 1. 现状分析

### 1.1 项目中的注册表散落情况

当前项目中有 10 个注册表相关实现，分布在 6 个项目中：

| # | 注册表 | 位置 | 模式 | 键 | 值 | 行数 |
|---|--------|------|------|-----|-----|------|
| 1 | `FunctionRegistry` | `CommandFramework/Execution` | 键值 | `string` | `Delegate` | 126 |
| 2 | `CommandRegistry` | `CommandFramework` | 键值 | `string` | `ICommand` | 122 |
| 3 | `ConfigContainer._subContainers` | `WdUI/UI` | 键值 | `string` | `ConfigContainer` | 80 |
| 4 | `LayoutStrategyRegistry` | `WdUI/Layout` | 键值 | `string` | `ILayoutStrategy` | 35 |
| 5 | `FunctionStorageRegistry` | `FunctionStorage` | 键值 | `string` | `Delegate` | 106 |
| 6 | `IDataSourceRegistry` | `FrameworkAbstractions` | 键值 | `string` | `T` | 34 |
| 7 | `StrategyRegistry<TKey,TIn,TOut>` | `Pipeline` | 键值+执行 | `TKey` | `IStrategy` | 111 |
| 8 | `PipelineStageRegistry` | `Pipeline` | 阶段 | `PipelineStage` | `IPipeline` | 447 |
| 9 | `PipelineStageRegistry<TContext>` | `CentralPipeline` | 阶段 | `PipelineStage` | `List<IPipeline>` | 348 |
| 10 | `IWorkflowRegistry` | `WorkFlowMethod` | 图 | `string` | - | 5 |

### 1.2 三种模式

**模式 A — 键值注册表**（7 个实例）：`Register(key, value)` → `Find(key)` → `Use(value)`

**模式 B — 阶段注册表**（2 个实例）：`Register(stage, value)` → `GetValues(stage)` → `ExecuteAll(values, input)`

**模式 C — 图注册表**（1 个实例）：`RegisterNode(node)` + `RegisterEdge(edge)` → `Traverse(graph)` → `Execute(node, context)`

### 1.3 问题

- 每个注册表独立实现字典、锁、事件通知，代码重复
- 接口不统一，上层代码无法统一处理注册表
- 新增注册表需要重新实现所有方法

### 2.3 基类

#### MapRegistryBase

```csharp
public class MapRegistryBase<TKey, TValue> : IMapRegistry<TKey, TValue>
    where TKey : notnull
{
    private readonly ConcurrentDictionary<TKey, TValue> _store;
    private readonly bool _allowOverwrite;

    public MapRegistryBase(IEqualityComparer<TKey>? comparer = null, bool allowOverwrite = true)
    {
        _store = new ConcurrentDictionary<TKey, TValue>(comparer ?? EqualityComparer<TKey>.Default);
        _allowOverwrite = allowOverwrite;
    }

    protected ConcurrentDictionary<TKey, TValue> Store => _store;

    protected virtual void OnBeforeRegister(TKey key, TValue value) { }
    protected virtual void OnBeforeUnregister(TKey key) { }
    protected virtual void OnBeforeClear() { }

    public virtual void Register(TKey key, TValue value)
    {
        ArgumentNullException.ThrowIfNull(key);
        ArgumentNullException.ThrowIfNull(value);
        OnBeforeRegister(key, value);
        if (!_allowOverwrite && _store.ContainsKey(key))
            throw new InvalidOperationException($"Key '{key}' is already registered.");
        _store[key] = value;
    }

    public bool TryGet(TKey key, out TValue? value)
    {
        ArgumentNullException.ThrowIfNull(key);
        return _store.TryGetValue(key, out value);
    }

    public bool Contains(TKey key)
    {
        ArgumentNullException.ThrowIfNull(key);
        return _store.ContainsKey(key);
    }

    public IEnumerable<TKey> GetKeys() => _store.Keys;

    public virtual void Unregister(TKey key)
    {
        ArgumentNullException.ThrowIfNull(key);
        OnBeforeUnregister(key);
        _store.TryRemove(key, out _);
    }

    public virtual void Clear()
    {
        OnBeforeClear();
        _store.Clear();
    }
}
```

#### StageRegistryBase

```csharp
public class StageRegistryBase<TStage, TValue> : IStageRegistry<TStage, TValue>
    where TStage : struct, Enum
{
    private readonly ConcurrentDictionary<TStage, List<TValue>> _store = new();
    private readonly ReaderWriterLockSlim _lock = new();

    public void Register(TStage stage, TValue value)
    {
        ArgumentNullException.ThrowIfNull(value);
        _lock.EnterWriteLock();
        try
        {
            if (!_store.ContainsKey(stage))
                _store[stage] = new List<TValue>();
            _store[stage].Add(value);
        }
        finally { _lock.ExitWriteLock(); }
    }

    public void InsertAt(TStage stage, int index, TValue value)
    {
        ArgumentNullException.ThrowIfNull(value);
        _lock.EnterWriteLock();
        try
        {
            if (!_store.ContainsKey(stage))
                _store[stage] = new List<TValue>();
            var list = _store[stage];
            index = Math.Clamp(index, 0, list.Count);
            list.Insert(index, value);
        }
        finally { _lock.ExitWriteLock(); }
    }

    public void Replace(TStage stage, int index, TValue value)
    {
        ArgumentNullException.ThrowIfNull(value);
        _lock.EnterWriteLock();
        try
        {
            if (_store.TryGetValue(stage, out var list) && index >= 0 && index < list.Count)
                list[index] = value;
        }
        finally { _lock.ExitWriteLock(); }
    }

    public void ReplaceAll(TStage stage, IEnumerable<TValue> values)
    {
        ArgumentNullException.ThrowIfNull(values);
        _lock.EnterWriteLock();
        try { _store[stage] = values.ToList(); }
        finally { _lock.ExitWriteLock(); }
    }

    public void Remove(TStage stage, int index)
    {
        _lock.EnterWriteLock();
        try
        {
            if (_store.TryGetValue(stage, out var list) && index >= 0 && index < list.Count)
                list.RemoveAt(index);
        }
        finally { _lock.ExitWriteLock(); }
    }

    public void ClearStage(TStage stage)
    {
        _lock.EnterWriteLock();
        try
        {
            if (_store.ContainsKey(stage))
                _store[stage] = new List<TValue>();
        }
        finally { _lock.ExitWriteLock(); }
    }

    public void Clear()
    {
        _lock.EnterWriteLock();
        try { _store.Clear(); }
        finally { _lock.ExitWriteLock(); }
    }

    public IReadOnlyList<TValue> GetValues(TStage stage)
    {
        _lock.EnterReadLock();
        try
        {
            return _store.TryGetValue(stage, out var list)
                ? list.ToList().AsReadOnly()
                : Array.Empty<TValue>();
        }
        finally { _lock.ExitReadLock(); }
    }

    public IReadOnlyDictionary<TStage, IReadOnlyList<TValue>> GetAllValues()
    {
        _lock.EnterReadLock();
        try
        {
            return _store.ToDictionary(
                kvp => kvp.Key,
                kvp => (IReadOnlyList<TValue>)kvp.Value.ToList().AsReadOnly());
        }
        finally { _lock.ExitReadLock(); }
    }

    public int GetCount(TStage stage)
    {
        _lock.EnterReadLock();
        try
        {
            return _store.TryGetValue(stage, out var list) ? list.Count : 0;
        }
        finally { _lock.ExitReadLock(); }
    }

    public IEnumerable<TStage> GetStages() => _store.Keys;

    public int TotalCount
    {
        get
        {
            _lock.EnterReadLock();
            try { return _store.Values.Sum(list => list.Count); }
            finally { _lock.ExitReadLock(); }
        }
    }
}
```

#### GraphRegistryBase

```csharp
public class GraphRegistryBase<TNodeId, TNode, TEdge> : IGraphRegistry<TNodeId, TNode, TEdge>
    where TNodeId : notnull
{
    private readonly ConcurrentDictionary<TNodeId, TNode> _nodes = new();
    private readonly ConcurrentDictionary<TNodeId, List<(TNodeId To, TEdge Edge)>> _outgoing = new();
    private readonly ConcurrentDictionary<TNodeId, List<(TNodeId From, TEdge Edge)>> _incoming = new();

    public void RegisterNode(TNodeId id, TNode node)
    {
        ArgumentNullException.ThrowIfNull(id);
        ArgumentNullException.ThrowIfNull(node);
        _nodes[id] = node;
    }

    public void RegisterEdge(TNodeId from, TNodeId to, TEdge edge)
    {
        ArgumentNullException.ThrowIfNull(from);
        ArgumentNullException.ThrowIfNull(to);
        ArgumentNullException.ThrowIfNull(edge);

        _outgoing.AddOrUpdate(from,
            _ => new List<(TNodeId, TEdge)> { (to, edge) },
            (_, list) => { list.Add((to, edge)); return list; });

        _incoming.AddOrUpdate(to,
            _ => new List<(TNodeId, TEdge)> { (from, edge) },
            (_, list) => { list.Add((from, edge)); return list; });
    }

    public TNode? GetNode(TNodeId id)
        => _nodes.TryGetValue(id, out var node) ? node : default;

    public IEnumerable<TEdge> GetOutgoingEdges(TNodeId nodeId)
        => _outgoing.TryGetValue(nodeId, out var edges)
            ? edges.Select(e => e.Edge) : Enumerable.Empty<TEdge>();

    public IEnumerable<TEdge> GetIncomingEdges(TNodeId nodeId)
        => _incoming.TryGetValue(nodeId, out var edges)
            ? edges.Select(e => e.Edge) : Enumerable.Empty<TEdge>();

    public void UnregisterNode(TNodeId id)
    {
        _nodes.TryRemove(id, out _);
        _outgoing.TryRemove(id, out _);
        _incoming.TryRemove(id, out _);
    }

    public void UnregisterEdge(TNodeId from, TNodeId to)
    {
        if (_outgoing.TryGetValue(from, out var outEdges))
        {
            outEdges.RemoveAll(e => e.To.Equals(to));
            if (outEdges.Count == 0) _outgoing.TryRemove(from, out _);
        }
        if (_incoming.TryGetValue(to, out var inEdges))
        {
            inEdges.RemoveAll(e => e.From.Equals(from));
            if (inEdges.Count == 0) _incoming.TryRemove(to, out _);
        }
    }

    public IEnumerable<TNode> GetAllNodes() => _nodes.Values;
    public IEnumerable<TEdge> GetAllEdges()
        => _outgoing.Values.SelectMany(list => list.Select(e => e.Edge));

    public IEnumerable<TNodeId> TopologicalSort(TNodeId startNodeId)
    {
        var visited = new HashSet<TNodeId>();
        var result = new List<TNodeId>();
        Dfs(startNodeId, visited, result);
        return result;
    }

    private void Dfs(TNodeId nodeId, HashSet<TNodeId> visited, List<TNodeId> result)
    {
        if (!visited.Add(nodeId)) return;
        if (_outgoing.TryGetValue(nodeId, out var edges))
        {
            foreach (var (to, _) in edges)
                Dfs(to, visited, result);
        }
        result.Add(nodeId);
    }

    public void Clear()
    {
        _nodes.Clear();
        _outgoing.Clear();
        _incoming.Clear();
    }
}
```

## 3. 接入方案

### 3.1 模式 A 接入

所有模式 A 注册表通过**显式接口实现**接入，不修改现有方法：

```csharp
// FunctionRegistry 示例
public class FunctionRegistry : IFunctionRegistry, IMapRegistry<string, Delegate>
{
    // 现有方法不变...

    void IMapRegistry<string, Delegate>.Register(string key, Delegate value) => Register(key, value);
    bool IMapRegistry<string, Delegate>.TryGet(string key, out Delegate? value)
    {
        if (_functions.TryGetValue(key, out var func)) { value = func.Delegate; return true; }
        value = null; return false;
    }
    bool IMapRegistry<string, Delegate>.Contains(string key) => _functions.ContainsKey(key);
    IEnumerable<string> IMapRegistry<string, Delegate>.GetKeys() => _functions.Keys;
    void IMapRegistry<string, Delegate>.Unregister(string key) => Unregister(key);
}
```

每个注册表改动量：+15~25 行，0 行修改。

### 3.2 模式 B 接入

```csharp
// PipelineStageRegistry 改造后
public class PipelineStageRegistry : StageRegistryBase<PipelineStage, IPipeline<object, object>>
{
    private IReadOnlyList<IPipeline<object, object>>? _cachedPipelines;

    public void Register(PipelineStage stage, IPipeline<object, object> pipeline)
        => base.Register(stage, pipeline);

    public IPipeline<object, object>? Get(PipelineStage stage)
        => GetValues(stage).FirstOrDefault();

    public IReadOnlyList<IPipeline<object, object>> GetAllPipelines()
    {
        if (_cachedPipelines != null) return _cachedPipelines;
        _cachedPipelines = PipelineStageExtensions.GetAllStages()
            .SelectMany(s => GetValues(s))
            .ToList().AsReadOnly();
        return _cachedPipelines;
    }

    public T ExecuteAll<T>(T input, IExecutionStrategy? strategy = null)
    {
        strategy ??= new SequentialStrategy();
        var pipelines = GetAllPipelines()
            .Select(p => new DelegatePipeline<T, T>(p.Name,
                sync: i => (T)p.Execute(i!),
                async: async i => (T)await p.ExecuteAsync(i!).ConfigureAwait(false)))
            .ToList() as IReadOnlyList<IPipeline<T, T>>;
        return strategy.Execute(pipelines, input);
    }
}
```

改动量：继承基类，删除重复字典和锁逻辑，约 -200 行。
